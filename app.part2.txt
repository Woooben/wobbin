  function dbOpen(){return new Promise((resolve,reject)=>{const r=indexedDB.open(DB_NAME,DB_VERSION);r.onupgradeneeded=()=>{const db=r.result;if(!db.objectStoreNames.contains(STORE))db.createObjectStore(STORE,{keyPath:'id'})};r.onsuccess=()=>resolve(r.result);r.onerror=()=>reject(r.error)})}
  async function dbAll(){const db=await dbOpen();return new Promise((resolve,reject)=>{const r=db.transaction(STORE,'readonly').objectStore(STORE).getAll();r.onsuccess=()=>resolve(r.result||[]);r.onerror=()=>reject(r.error)})}
  async function dbPut(item){const db=await dbOpen();return new Promise((resolve,reject)=>{const r=db.transaction(STORE,'readwrite').objectStore(STORE).put(item);r.onsuccess=()=>resolve();r.onerror=()=>reject(r.error)})}
  async function dbDelete(id){const db=await dbOpen();return new Promise((resolve,reject)=>{const r=db.transaction(STORE,'readwrite').objectStore(STORE).delete(id);r.onsuccess=()=>resolve();r.onerror=()=>reject(r.error)})}
  async function dbClear(){const db=await dbOpen();return new Promise((resolve,reject)=>{const r=db.transaction(STORE,'readwrite').objectStore(STORE).clear();r.onsuccess=()=>resolve();r.onerror=()=>reject(r.error)})}
  function hydrate(item){if(item.blob&&!item.imageUrl)item.imageUrl=URL.createObjectURL(item.blob);return item}
  async function init(){try{S.items=(await dbAll()).map(hydrate)}catch(e){toast('浏览器存储初始化失败')}finally{S.loading=false;render()}}

  function inferElements(text){const t=slug(text);return ELEMENTS.filter(e=>ELEM_KEYS[e].some(k=>t.includes(slug(k))))}
  function inferCategory(text){const t=slug(text);for(const [cat,keys] of Object.entries(CATEGORY_KEYS)){if(keys.some(k=>t.includes(slug(k))))return cat}return 'Other'}
  function inferPlatform(text,fallback='iOS'){const t=slug(text);if(t.includes('android'))return'Android';if(t.includes('web')||t.includes('desktop')||t.includes('site'))return'Web';if(t.includes('ios')||t.includes('iphone'))return'iOS';return fallback}
